function updateClock() { const clockElement = document.getElementById('clock'); if (!clockElement) { return; } const now = new Date(); const formatted = now.toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit', second: '2-digit' }); clockElement.textContent = formatted; } document.addEventListener('DOMContentLoaded', function () { updateClock(); setInterval(updateClock, 11000); });
